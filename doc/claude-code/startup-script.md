# cleanup script

Configure the user interface to not attribute commits, PRs or anything for that matter to Claude.
The web application still attaches a note to the PR despite the configuration blocking it, so we use some hooks to work around that.

```bash
#!/bin/bash

mkdir -p /root/.claude

cat > /root/.claude/settings.json <<'EOF'
{
  "attribution": {
    "commit": "",
    "pr": "",
    "sessionUrl": false
  },
  "autoCompactWindow": 300000,
  "hooks": {
    "Stop": [
      {
        "hooks": [
          {
            "type": "command",
            "command": "/root/.claude/clean-pr-attribution.sh"
          }
        ]
      }
    ],
    "PostToolUse": [
      {
        "matcher": "mcp__github__create_pull_request",
        "hooks": [{ "type": "command", "command": "/root/.claude/pr-attribution-pending.sh" }]
      }
    ]
  }
}
EOF

cat > "/root/.claude/pr-attribution-pending.sh" <<'EOF'
#!/bin/bash

INPUT=$(cat)

TOOL_NAME=$(echo "$INPUT" | grep -o '"tool_name"[[:space:]]*:[[:space:]]*"[^"]*"' | sed -E 's/.*"([^"]+)"$/\1/')

if [ "$TOOL_NAME" != "mcp__github__create_pull_request" ]; then
  exit 0
fi

PR_NUMBER=$(echo "$INPUT" | grep -o '"url"[[:space:]]*:[[:space:]]*"[^"]*/pull/[0-9]\+"' | grep -o '[0-9]\+"$' | tr -d '"')

if [ -z "$PR_NUMBER" ]; then
  exit 0
fi

mkdir -p /root/.claude/pr-attribution-pending
touch "/root/.claude/pr-attribution-pending/$PR_NUMBER"
EOF

chmod +x "/root/.claude/pr-attribution-pending.sh"


cat > "/root/.claude/clean-pr-attribution.sh" <<'EOF'
#!/bin/bash

PENDING_DIR="/root/.claude/pr-attribution-pending"
mkdir -p "$PENDING_DIR"

PENDING=$(ls "$PENDING_DIR" 2>/dev/null)

if [ -z "$PENDING" ]; then
  exit 0
fi

PR_LIST=$(echo "$PENDING" | tr '\n' ' ')

cat <<EOF
{"decision": "block", "reason": "These PR numbers have pending attribution cleanup: $PR_LIST. For each one, inspect its body and remove any Claude Code attribution (claude.ai/code/session_* URLs, \"Generated by Claude Code\", and the associated separator). Do not modify any other PR content. After cleaning each PR, delete its marker file from $PENDING_DIR."}
EOF

chmod +x "/root/.claude/clean-pr-attribution.sh"
```
